from Cryptodome.Cipher import AES 
from Cryptodome.Util.Padding import unpad 
import sqlite3 
import base64 
 
# AES 解密所需的密钥和初始化向量 
key = b'2024DataSecurity'  # 16 字节密钥 
iv = b'2024' + (b'\0' * 12)  # 16 字节IV，'2024' 后面跟了12 个空字符 
 
def aes_decrypt(encrypted_data, key, iv): 
    encrypted_data_bytes = base64.b64decode(encrypted_data) 
    cipher = AES.new(key, AES.MODE_CBC, iv) 
    decrypted_data_bytes = unpad(cipher.decrypt(encrypted_data_bytes), AES.block_size) 
    return decrypted_data_bytes.decode('utf-8') 
 
# 连接到SQLite 数据库 
conn = sqlite3.connect('output.db')  # 替换为实际数据库文件路径 
cursor = conn.cursor() 
 
# 查询所有加密的记录 
cursor.execute("SELECT id, usermobile, useremail, sfz FROM ay_member") 
rows = cursor.fetchall() 
 
# 解密并更新记录 
for row in rows: 
    id, encrypted_usermobile, encrypted_useremail, encrypted_sfz = row 
    decrypted_usermobile = aes_decrypt(encrypted_usermobile, key, iv) 
    decrypted_useremail = aes_decrypt(encrypted_useremail, key, iv) 
    decrypted_sfz = aes_decrypt(encrypted_sfz, key, iv) 
     
    cursor.execute(""" 
        UPDATE ay_member 
        SET usermobile = ?, useremail = ?, sfz = ? 
        WHERE id = ? 
    """, (decrypted_usermobile, decrypted_useremail, decrypted_sfz, id)) 
 
conn.commit() 
cursor.close() 
conn.close() 
 
print("解密更新完成。")

# 校验码计算函数 
def calculate_check_digit(id17): 
    # 系数列表 
    coefficients = [7, 9, 10, 5, 8, 4, 2, 1, 6, 3, 7, 9, 10, 5, 8, 4, 2] 
    # 校验码对应值 
    check_digit_map = {0: '1', 1: '0', 2: 'X', 3: '9', 4: '8', 5: '7', 6: '6', 7: '5', 8: '4', 9: '3', 10: '2'} 
     
    sum = 0
